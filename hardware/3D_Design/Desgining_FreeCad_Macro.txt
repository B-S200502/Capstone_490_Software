# ==========================================================
# Helpers
# ==========================================================
def clamp(v, lo, hi):
    return max(lo, min(hi, v))

def remove_splitters(shape):
    try:
        return shape.removeSplitter()
    except:
        return shape

def get_obj(key):
    o = doc.getObject(key)
    if o:
        return o
    try:
        hits = doc.getObjectsByLabel(key)
        if hits:
            return hits[0]
    except:
        pass
    k = key.lower()
    for ob in doc.Objects:
        if (ob.Name or "").lower() == k or (getattr(ob, "Label", "") or "").lower() == k:
            return ob
    return None

def ensure_feature(name):
    """Return (obj, created_bool)."""
    o = doc.getObject(name)
    created = False
    if o is None:
        o = doc.addObject("Part::Feature", name)
        o.Label = name
        created = True
    return o, created

def bb_center(bb):
    return (0.5*(bb.XMin+bb.XMax), 0.5*(bb.YMin+bb.YMax), 0.5*(bb.ZMin+bb.ZMax))

def safe_normalized(v):
    try:
        vv = App.Vector(v.x, v.y, v.z)
        if vv.Length < 1e-9:
            return vv
        vv.normalize()
        return vv
    except:
        return App.Vector(0,0,0)

def rounded_rect_face_xz(xL, xR, zB, zT, r, y_at):
    w = xR - xL
    h = zT - zB
    r = max(0.0, min(r, w/2.0, h/2.0))
    if r < 1e-6:
        p1 = App.Vector(xL, y_at, zB); p2 = App.Vector(xR, y_at, zB)
        p3 = App.Vector(xR, y_at, zT); p4 = App.Vector(xL, y_at, zT)
        wire = Part.Wire([Part.makeLine(p1,p2),Part.makeLine(p2,p3),Part.makeLine(p3,p4),Part.makeLine(p4,p1)])
        return Part.Face(wire)

    rt = r / math.sqrt(2.0)
    topL = App.Vector(xL + r, y_at, zT); topR = App.Vector(xR - r, y_at, zT)
    rightT = App.Vector(xR, y_at, zT - r); rightB = App.Vector(xR, y_at, zB + r)
    botR = App.Vector(xR - r, y_at, zB); botL = App.Vector(xL + r, y_at, zB)
    leftB = App.Vector(xL, y_at, zB + r); leftT = App.Vector(xL, y_at, zT - r)

    tr_mid = App.Vector((xR - r) + rt, y_at, (zT - r) + rt)
    br_mid = App.Vector((xR - r) + rt, y_at, (zB + r) - rt)
    bl_mid = App.Vector((xL + r) - rt, y_at, (zB + r) - rt)
    tl_mid = App.Vector((xL + r) - rt, y_at, (zT - r) + rt)

    edges = [
        Part.makeLine(topL, topR),
        Part.Arc(topR, tr_mid, rightT).toShape(),
        Part.makeLine(rightT, rightB),
        Part.Arc(rightB, br_mid, botR).toShape(),
        Part.makeLine(botR, botL),
        Part.Arc(botL, bl_mid, leftB).toShape(),
        Part.makeLine(leftB, leftT),
        Part.Arc(leftT, tl_mid, topL).toShape(),
    ]
    return Part.Face(Part.Wire(edges))

def find_front_glass_face_bbox(shape, tol=0.9):
    bb = shape.BoundBox
    y_front = bb.YMin
    best = None
    best_area = -1.0
    y_axis = App.Vector(0,-1,0)
    for f in shape.Faces:
        fbb = f.BoundBox
        if abs(fbb.YMin - y_front) > tol:
            continue
        try:
            n = safe_normalized(f.normalAt(0,0))
            if abs(n.dot(y_axis)) < 0.85:
                continue
            area = f.Area
            if area > best_area:
                best_area = area
                best = f
        except:
            continue
    if best:
        return best.BoundBox, True
    return bb, False

def shape_world(obj):
    sh = obj.Shape.copy()
    try:
        sh.Placement = obj.Placement
    except:
        pass
    return sh

# ==========================================================
# Grab key objects
# ==========================================================
pi   = get_obj("Pi5")
aud  = get_obj("AudioBoard")
pwr  = get_obj("PowerBoard")
cam  = get_obj("PiCamEnv")
lcd  = get_obj("LCD")

if not (pi and aud and pwr and cam and lcd):
    raise RuntimeError("Missing one of: Pi5, AudioBoard, PowerBoard, PiCamEnv, LCD")

hf0 = get_obj("Housing_Front")
hb0 = get_obj("Housing_Back")

# ==========================================================
# Reference center from existing housing
# ==========================================================
bb_ref = None
if hf0 and hasattr(hf0,"Shape") and not hf0.Shape.isNull():
    bb_ref = hf0.Shape.BoundBox
    if hb0 and hasattr(hb0,"Shape") and not hb0.Shape.isNull():
        bb_ref.add(hb0.Shape.BoundBox)

if bb_ref:
    hx, _, hz = bb_center(bb_ref)
else:
    hx, hz = 0.0, 0.0

LEFT_X   = hx - OUTER_W/2.0
RIGHT_X  = hx + OUTER_W/2.0
BOTTOM_Z = hz - OUTER_H/2.0
TOP_Z    = hz + OUTER_H/2.0
FRONT_Y  = -OUTER_T/2.0
BACK_Y   =  OUTER_T/2.0

inner_front_y = FRONT_Y + WALL
inner_back_y  = BACK_Y  - WALL

# ==========================================================
# LCD: center + flush
# ==========================================================
lcd_shape = lcd.Shape
lcd_glass_bb, used_glass = find_front_glass_face_bbox(lcd_shape, tol=0.9)
lcgx, lcgy, lcgz = bb_center(lcd_glass_bb)

if MOVE_LCD_CENTER_XZ:
    dx = hx - lcgx
    dz = hz - lcgz
    lcd.Placement.Base = lcd.Placement.Base.add(App.Vector(dx, 0, dz))
    lcd_shape = lcd.Shape
    lcd_glass_bb, used_glass = find_front_glass_face_bbox(lcd_shape, tol=0.9)
    lcgx, lcgy, lcgz = bb_center(lcd_glass_bb)

if MOVE_LCD_FLUSH_Y:
    desired = FRONT_Y + LCD_FLUSH_RECESS_Y
    dy = desired - lcd_glass_bb.YMin
    lcd.Placement.Base = lcd.Placement.Base.add(App.Vector(0, dy, 0))
    lcd_shape = lcd.Shape
    lcd_glass_bb, used_glass = find_front_glass_face_bbox(lcd_shape, tol=0.9)

# ==========================================================
# Build housing shell
# ==========================================================
outer_face = rounded_rect_face_xz(LEFT_X, RIGHT_X, BOTTOM_Z, TOP_Z, CORNER_R, FRONT_Y)
outer_solid = outer_face.extrude(App.Vector(0, OUTER_T, 0))

inner_face = rounded_rect_face_xz(
    LEFT_X + WALL, RIGHT_X - WALL,
    BOTTOM_Z + FLOOR, TOP_Z - LID,
    max(0.0, CORNER_R - WALL),
    inner_front_y
)
inner_solid = inner_face.extrude(App.Vector(0, OUTER_T - 2*WALL, 0))

shell = remove_splitters(outer_solid.cut(inner_solid))

# ==========================================================
# LCD opening + pocket (fit-friendly)
# ==========================================================
lcd_bb = lcd.Shape.BoundBox
glass = lcd_glass_bb

wx1 = glass.XMin - LCD_OPEN_X_CLEAR
wx2 = glass.XMax + LCD_OPEN_X_CLEAR
wz1 = glass.ZMin - LCD_OPEN_Z_BOTTOM_CLEAR
wz2 = glass.ZMax + LCD_OPEN_Z_TOP_CLEAR

wx1 = clamp(wx1, LEFT_X + 1.0, RIGHT_X - 1.0)
wx2 = clamp(wx2, LEFT_X + 1.0, RIGHT_X - 1.0)
wz1 = clamp(wz1, BOTTOM_Z + 1.0, TOP_Z - 1.0)
wz2 = clamp(wz2, BOTTOM_Z + 1.0, TOP_Z - 1.0)

win_w = max(1.0, wx2 - wx1)
win_h = max(1.0, wz2 - wz1)

win_depth = WALL + EXTRA_CUT
win = Part.makeBox(win_w, win_depth, win_h, App.Vector(wx1, FRONT_Y - 0.5, wz1))
shell = shell.cut(win)

px1 = lcd_bb.XMin - LCD_POCKET_CLEAR
px2 = lcd_bb.XMax + LCD_POCKET_CLEAR
pz1 = lcd_bb.ZMin - LCD_POCKET_CLEAR
pz2 = lcd_bb.ZMax + LCD_POCKET_CLEAR

px1 = clamp(px1, LEFT_X + 1.0, RIGHT_X - 1.0)
px2 = clamp(px2, LEFT_X + 1.0, RIGHT_X - 1.0)
pz1 = clamp(pz1, BOTTOM_Z + 1.0, TOP_Z - 1.0)
pz2 = clamp(pz2, BOTTOM_Z + 1.0, TOP_Z - 1.0)

pocket_start_y = FRONT_Y + LCD_WALL_SKIN
pocket_depth = (lcd_bb.YMax - pocket_start_y) + LCD_POCKET_EXTRA_DEPTH
pocket_depth = max(2.0, pocket_depth)
pocket_depth = min((inner_back_y - pocket_start_y) - 1.0, pocket_depth)

pocket = Part.makeBox(max(1.0, px2-px1), pocket_depth, max(1.0, pz2-pz1),
                      App.Vector(px1, pocket_start_y, pz1))
shell = shell.cut(pocket)
shell = remove_splitters(shell)

# ==========================================================
# Split into halves
# ==========================================================
def make_keep(y0, y1):
    return Part.makeBox(OUTER_W + 300, y1-y0, OUTER_H + 300,
                        App.Vector(LEFT_X-150, y0, BOTTOM_Z-150))

front_keep = make_keep(FRONT_Y-1.0, SPLIT_Y)
back_keep  = make_keep(SPLIT_Y, BACK_Y+1.0)

front_shape = remove_splitters(shell.common(front_keep))
back_shape  = remove_splitters(shell.common(back_keep))

# ==========================================================
# BACK HALF: camera hole
# ==========================================================
CAM_HOLE_D = 14.0
cam_bb = cam.Shape.BoundBox
cam_cx, cam_cy, cam_cz = bb_center(cam_bb)
cam_len = (BACK_Y - SPLIT_Y) + EXTRA_CUT
cam_hole = Part.makeCylinder(CAM_HOLE_D/2.0, cam_len,
                             App.Vector(cam_cx, BACK_Y + 0.5, cam_cz),
                             App.Vector(0,-1,0))
back_shape = back_shape.cut(cam_hole)

# ==========================================================
# BACK HALF: MicPattern_Helper (editable) — NO centering anymore
# ==========================================================
mic_helper, mic_created = ensure_feature("MicPattern_Helper")

mic_y0 = SPLIT_Y - 2.0
mic_len = (BACK_Y - mic_y0) + MIC_LEN_EXTRA
cyls = []
for _, mx, mz in MICS:
    cyls.append(
        Part.makeCylinder(MIC_HOLE_D/2.0, mic_len,
                          App.Vector(mx, mic_y0, mz),
                          App.Vector(0,1,0))
    )
mic_helper.Shape = remove_splitters(Part.makeCompound(cyls))

# only set initial placement when created (never overwrite your manual moves)
if mic_created:
    ab_bb = aud.Shape.BoundBox
    ab_cx, ab_cy, ab_cz = bb_center(ab_bb)
    mic_helper.Placement = App.Placement(
        App.Vector(ab_cx, 0.0, ab_cz),
        App.Rotation(App.Vector(0,1,0), MIC_ROT_DEG_INIT)
    )

back_shape = back_shape.cut(shape_world(mic_helper))

# ==========================================================
# BACK HALF: USB-A cutter helper (editable, REAL hole)
# ==========================================================
raspi_cut, raspi_created = ensure_feature("RasPi_USB_A_Cutout")

# Make the cutter in local coords:
# X = width, Y = depth, Z = height  (depth goes through the wall)
raspi_cut.Shape = Part.makeBox(
    USB_A_W2, USB_A_DEPTH, USB_A_H,
    App.Vector(-USB_A_W2/2.0, -USB_A_DEPTH, -USB_A_H/2.0)
)

# initial placement only when created (preserve your current "perfect" position otherwise)
if raspi_created:
    pi_bb = pi.Shape.BoundBox
    px, py, pz = bb_center(pi_bb)
    y_center = clamp(py, SPLIT_Y + 8.0, BACK_Y - 8.0)
    # start it on the BACK outside face by default (you can move it)
    raspi_cut.Placement = App.Placement(App.Vector(px, BACK_Y, pz), App.Rotation())

# keep cutter inside Housing_Back region (so it never reaches Housing_Front)
b = raspi_cut.Placement.Base
raspi_cut.Placement.Base = App.Vector(b.x, clamp(b.y, SPLIT_Y + 6.0, BACK_Y), b.z)

back_shape = back_shape.cut(shape_world(raspi_cut))
back_shape = remove_splitters(back_shape)

# ==========================================================
# FRONT HALF: USB-C cutter helper (editable, can go lower)
# ==========================================================
usb_c_cut, usb_c_created = ensure_feature("Power_USB_C_Cutout")

usb_c_cut.Shape = Part.makeBox(
    USB_C_W, USB_C_DEPTH, USB_C_H,
    App.Vector(-USB_C_W/2.0, 0.0, -USB_C_H/2.0)
)

if usb_c_created:
    pbb = pwr.Shape.BoundBox
    pcx, pcy, pcz = bb_center(pbb)
    x0 = clamp(pcx, LEFT_X + USB_C_SAFE_EDGE_X, RIGHT_X - USB_C_SAFE_EDGE_X)
    z0 = clamp(pcz, BOTTOM_Z + USB_C_SAFE_EDGE_Z, TOP_Z - USB_C_SAFE_EDGE_Z)
    usb_c_cut.Placement = App.Placement(App.Vector(x0, FRONT_Y, z0), App.Rotation())

# Allow lower movement: only clamp Z lightly now; never force X
bb = usb_c_cut.Placement.Base
usb_c_cut.Placement.Base = App.Vector(
    bb.x,
    FRONT_Y,
    clamp(bb.z, BOTTOM_Z + USB_C_SAFE_EDGE_Z, TOP_Z - USB_C_SAFE_EDGE_Z)
)

front_shape = front_shape.cut(shape_world(usb_c_cut))
front_shape = remove_splitters(front_shape)

# ==========================================================
# Write back
# ==========================================================
hf, _ = ensure_feature("Housing_Front")
hb, _ = ensure_feature("Housing_Back")
hf.Shape = front_shape
hb.Shape = back_shape

doc.recompute()

print("\n=== Rebuild done (no fan edits) ===")
print("Helpers you can edit in GUI:")
print(" - RasPi_USB_A_Cutout (real hole in Housing_Back only)")
print(" - Power_USB_C_Cutout (move lower in Z if needed)")
print(" - MicPattern_Helper (move/rotate until it overlaps PCB mic holes)")
print(f"USB-C Z clamp: bottom edge = {USB_C_SAFE_EDGE_Z}mm (lower this if you still need lower)")
