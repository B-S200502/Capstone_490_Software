import FreeCAD as App
import Part, math, re

doc = App.ActiveDocument
if not doc:
    raise RuntimeError("Open your .FCStd first (must have an ActiveDocument).")

# ==========================================================
# HOUSING DIMENSIONS (match your design)
# ==========================================================
OUTER_W = 250.0
OUTER_T = 80.0
OUTER_H = 140.0

WALL  = 2.6
FLOOR = 3.0
LID   = 2.6
CORNER_R = 22.0

# Split line between front/back (same meaning as before)
SPLIT_Y = 5.0

# Small extra depth for cutters so they fully cut (safe)
EXTRA_CUT = 6.0

# ==========================================================
# LCD WINDOW / POCKET SETTINGS
# ==========================================================
# Window clearance around LCD glass (in X and Z)
LCD_OPEN_X_CLEAR = 0.70
LCD_OPEN_Z_TOP_CLEAR = 0.70
LCD_OPEN_Z_BOTTOM_CLEAR = 0.10

# Pocket clearance around entire LCD envelope
LCD_POCKET_CLEAR = 1.10
# Extra depth behind LCD (fit-friendly)
LCD_POCKET_EXTRA_DEPTH = 2.0

# IMPORTANT: Make front wall thickness uniform (NO thinning)
# Pocket starts at the inner wall, not closer to the outside.
# (So outer wall thickness stays WALL everywhere.)
# ==========================================================

# ==========================================================
# Helpers
# ==========================================================
def clamp(v, lo, hi):
    return max(lo, min(hi, v))

def remove_splitters(shape):
    try:
        return shape.removeSplitter()
    except:
        return shape

def get_obj(key):
    o = doc.getObject(key)
    if o:
        return o
    try:
        hits = doc.getObjectsByLabel(key)
        if hits:
            return hits[0]
    except:
        pass
    k = key.lower()
    for ob in doc.Objects:
        if (ob.Name or "").lower() == k or (getattr(ob, "Label", "") or "").lower() == k:
            return ob
    return None

def ensure_feature(name):
    o = doc.getObject(name)
    if o is None:
        o = doc.addObject("Part::Feature", name)
        o.Label = name
    return o

def bb_center(bb):
    return (0.5*(bb.XMin+bb.XMax), 0.5*(bb.YMin+bb.YMax), 0.5*(bb.ZMin+bb.ZMax))

def safe_normalized(v):
    try:
        vv = App.Vector(v.x, v.y, v.z)
        if vv.Length < 1e-9:
            return vv
        vv.normalize()
        return vv
    except:
        return App.Vector(0,0,0)

def shape_world(obj):
    sh = obj.Shape.copy()
    sh.Placement = obj.Placement
    return sh

def rounded_rect_face_xz(xL, xR, zB, zT, r, y_at):
    w = xR - xL
    h = zT - zB
    r = max(0.0, min(r, w/2.0, h/2.0))
    if r < 1e-6:
        p1 = App.Vector(xL, y_at, zB); p2 = App.Vector(xR, y_at, zB)
        p3 = App.Vector(xR, y_at, zT); p4 = App.Vector(xL, y_at, zT)
        wire = Part.Wire([Part.makeLine(p1,p2),Part.makeLine(p2,p3),Part.makeLine(p3,p4),Part.makeLine(p4,p1)])
        return Part.Face(wire)

    rt = r / math.sqrt(2.0)
    topL = App.Vector(xL + r, y_at, zT); topR = App.Vector(xR - r, y_at, zT)
    rightT = App.Vector(xR, y_at, zT - r); rightB = App.Vector(xR, y_at, zB + r)
    botR = App.Vector(xR - r, y_at, zB); botL = App.Vector(xL + r, y_at, zB)
    leftB = App.Vector(xL, y_at, zB + r); leftT = App.Vector(xL, y_at, zT - r)

    tr_mid = App.Vector((xR - r) + rt, y_at, (zT - r) + rt)
    br_mid = App.Vector((xR - r) + rt, y_at, (zB + r) - rt)
    bl_mid = App.Vector((xL + r) - rt, y_at, (zB + r) - rt)
    tl_mid = App.Vector((xL + r) - rt, y_at, (zT - r) + rt)

    edges = [
        Part.makeLine(topL, topR),
        Part.Arc(topR, tr_mid, rightT).toShape(),
        Part.makeLine(rightT, rightB),
        Part.Arc(rightB, br_mid, botR).toShape(),
        Part.makeLine(botR, botL),
        Part.Arc(botL, bl_mid, leftB).toShape(),
        Part.makeLine(leftB, leftT),
        Part.Arc(leftT, tl_mid, topL).toShape(),
    ]
    return Part.Face(Part.Wire(edges))

def keep_box(xL, xR, zB, zT, y0, y1):
    return Part.makeBox((xR-xL)+300, (y1-y0), (zT-zB)+300,
                        App.Vector(xL-150, y0, zB-150))

def find_front_glass_face_bbox_world(shape_worldspace, tol=1.0):
    """
    Try to find the biggest face on the *front-most* side (smallest Y),
    with normal roughly pointing -Y. Return its bbox.
    Fallback: bbox of whole LCD.
    """
    bb = shape_worldspace.BoundBox
    y_front = bb.YMin
    best = None
    best_area = -1.0
    y_axis = App.Vector(0,-1,0)

    for f in shape_worldspace.Faces:
        fbb = f.BoundBox
        if abs(fbb.YMin - y_front) > tol:
            continue
        try:
            n = safe_normalized(f.normalAt(0,0))
            if abs(n.dot(y_axis)) < 0.85:
                continue
            if f.Area > best_area:
                best_area = f.Area
                best = f
        except:
            continue

    if best:
        return best.BoundBox
    return bb

# ==========================================================
# Grab LCD (required)
# ==========================================================
lcd = get_obj("LCD")
if not lcd:
    raise RuntimeError("Missing object: LCD")

# Existing housings (only used to keep center stable if they exist)
hf0 = get_obj("Housing_Front")
hb0 = get_obj("Housing_Back")

# ==========================================================
# Reference center (stable)
# ==========================================================
bb_ref = None
if hf0 and hasattr(hf0,"Shape") and not hf0.Shape.isNull():
    bb_ref = hf0.Shape.BoundBox
    if hb0 and hasattr(hb0,"Shape") and not hb0.Shape.isNull():
        bb_ref.add(hb0.Shape.BoundBox)

if bb_ref:
    hx, _, hz = bb_center(bb_ref)
else:
    hx, hz = 0.0, 0.0

LEFT_X   = hx - OUTER_W/2.0
RIGHT_X  = hx + OUTER_W/2.0
BOTTOM_Z = hz - OUTER_H/2.0
TOP_Z    = hz + OUTER_H/2.0
FRONT_Y  = -OUTER_T/2.0
BACK_Y   =  OUTER_T/2.0

inner_front_y = FRONT_Y + WALL
inner_back_y  = BACK_Y  - WALL

# ==========================================================
# Build CLEAN housing shell (NO holes at all)
# ==========================================================
outer_face = rounded_rect_face_xz(LEFT_X, RIGHT_X, BOTTOM_Z, TOP_Z, CORNER_R, FRONT_Y)
outer_solid = outer_face.extrude(App.Vector(0, OUTER_T, 0))

inner_face = rounded_rect_face_xz(
    LEFT_X + WALL, RIGHT_X - WALL,
    BOTTOM_Z + FLOOR, TOP_Z - LID,
    max(0.0, CORNER_R - WALL),
    inner_front_y
)
inner_solid = inner_face.extrude(App.Vector(0, OUTER_T - 2*WALL, 0))

shell = remove_splitters(outer_solid.cut(inner_solid))

# ==========================================================
# LCD window + pocket (AUTO-aligned to CURRENT LCD placement)
# ==========================================================
lcd_world = shape_world(lcd)
glass_bb = find_front_glass_face_bbox_world(lcd_world, tol=1.0)
lcd_bb = lcd_world.BoundBox

# Window extents around glass (X/Z)
wx1 = glass_bb.XMin - LCD_OPEN_X_CLEAR
wx2 = glass_bb.XMax + LCD_OPEN_X_CLEAR
wz1 = glass_bb.ZMin - LCD_OPEN_Z_BOTTOM_CLEAR
wz2 = glass_bb.ZMax + LCD_OPEN_Z_TOP_CLEAR

# keep within outer bounds
wx1 = clamp(wx1, LEFT_X + 1.0, RIGHT_X - 1.0)
wx2 = clamp(wx2, LEFT_X + 1.0, RIGHT_X - 1.0)
wz1 = clamp(wz1, BOTTOM_Z + 1.0, TOP_Z - 1.0)
wz2 = clamp(wz2, BOTTOM_Z + 1.0, TOP_Z - 1.0)

win_w = max(1.0, wx2 - wx1)
win_h = max(1.0, wz2 - wz1)
win_depth = WALL + EXTRA_CUT

# Cut window through the front wall only (but overshoot is fine)
win = Part.makeBox(win_w, win_depth, win_h, App.Vector(wx1, FRONT_Y - 0.5, wz1))
shell = remove_splitters(shell.cut(win))

# Pocket (starts at inner wall => keeps front thickness uniform)
px1 = lcd_bb.XMin - LCD_POCKET_CLEAR
px2 = lcd_bb.XMax + LCD_POCKET_CLEAR
pz1 = lcd_bb.ZMin - LCD_POCKET_CLEAR
pz2 = lcd_bb.ZMax + LCD_POCKET_CLEAR

px1 = clamp(px1, LEFT_X + 1.0, RIGHT_X - 1.0)
px2 = clamp(px2, LEFT_X + 1.0, RIGHT_X - 1.0)
pz1 = clamp(pz1, BOTTOM_Z + 1.0, TOP_Z - 1.0)
pz2 = clamp(pz2, BOTTOM_Z + 1.0, TOP_Z - 1.0)

pocket_start_y = inner_front_y  # <- key change (no thinning)
pocket_depth = (lcd_bb.YMax - pocket_start_y) + LCD_POCKET_EXTRA_DEPTH

max_depth = (inner_back_y - pocket_start_y) - 1.0
if max_depth < 2.0:
    max_depth = 2.0
pocket_depth = clamp(pocket_depth, 2.0, max_depth)

pocket = Part.makeBox(max(1.0, px2-px1), pocket_depth, max(1.0, pz2-pz1),
                      App.Vector(px1, pocket_start_y, pz1))
shell = remove_splitters(shell.cut(pocket))

# ==========================================================
# Split into halves (Front/Back)
# ==========================================================
front_keep = keep_box(LEFT_X, RIGHT_X, BOTTOM_Z, TOP_Z, FRONT_Y-1.0, SPLIT_Y)
back_keep  = keep_box(LEFT_X, RIGHT_X, BOTTOM_Z, TOP_Z, SPLIT_Y, BACK_Y+1.0)

front_shape = remove_splitters(shell.common(front_keep))
back_shape  = remove_splitters(shell.common(back_keep))

# ==========================================================
# Write back
# ==========================================================
hf = ensure_feature("Housing_Front")
hb = ensure_feature("Housing_Back")
hf.Shape = front_shape
hb.Shape = back_shape

doc.recompute()

print("\n=== DONE: CLEAN HOUSING (LCD window only) ===")
print("All other holes (mic/usb/camera/etc.) are removed/plugged by rebuild.")
print("LCD window auto-aligned to your CURRENT LCD placement.")
print("Front wall thickness stays uniform (no LCD thinning).")
print("\nEdit size here if needed:")
print(" - LCD_OPEN_* for the window")
print(" - LCD_POCKET_* for the internal pocket")
