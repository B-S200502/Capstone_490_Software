# ==========================================================
# EASY OBJECT GETTERS
# ==========================================================
def get_obj(name_or_label):
    o = doc.getObject(name_or_label)
    if o:
        return o
    try:
        hits = doc.getObjectsByLabel(name_or_label)
        if hits:
            return hits[0]
    except:
        pass
    key = name_or_label.lower()
    for ob in doc.Objects:
        if (ob.Name or "").lower() == key or (getattr(ob, "Label", "") or "").lower() == key:
            return ob
    return None

def ensure_feature(name):
    """
    Finds by Name first, then by Label, otherwise creates.
    This prevents duplicate 'HoldWall_L04001' surprises.
    """
    o = doc.getObject(name)
    if o:
        return o, False
    o = get_obj(name)
    if o:
        return o, False
    o = doc.addObject("Part::Feature", name)
    o.Label = name
    return o, True

def remove_splitters(shape):
    try:
        return shape.removeSplitter()
    except:
        return shape

def bb_center(bb):
    return (0.5*(bb.XMin+bb.XMax), 0.5*(bb.YMin+bb.YMax), 0.5*(bb.ZMin+bb.ZMax))

def shape_world(obj):
    sh = obj.Shape.copy()
    try:
        sh.Placement = obj.Placement
    except:
        pass
    return sh

def bbox_intersects(a, b):
    return not (
        a.XMax < b.XMin or a.XMin > b.XMax or
        a.YMax < b.YMin or a.YMin > b.YMax or
        a.ZMax < b.ZMin or a.ZMin > b.ZMax
    )

def find_front_glass_face_bbox(shape, tol=0.9):
    bb = shape.BoundBox
    y_front = bb.YMin
    best = None
    best_area = -1.0
    y_axis = App.Vector(0,-1,0)

    for f in shape.Faces:
        fbb = f.BoundBox
        if abs(fbb.YMin - y_front) > tol:
            continue
        try:
            n = f.normalAt(0,0)
            if n.Length < 1e-9:
                continue
            n.normalize()
            if abs(n.dot(y_axis)) < 0.85:
                continue
            if f.Area > best_area:
                best_area = f.Area
                best = f
        except:
            continue

    if best:
        return best.BoundBox, True
    return bb, False

def apply_gui_placement(obj, placement_dict):
    """
    placement_dict = {
        "angle": deg,
        "axis": (ax,ay,az),
        "pos":  (x,y,z)
    }
    """
    ang = float(placement_dict.get("angle", 0.0))
    ax, ay, az = placement_dict.get("axis", (0.0, 0.0, 1.0))
    px, py, pz = placement_dict.get("pos",  (0.0, 0.0, 0.0))

    axis_v = App.Vector(float(ax), float(ay), float(az))
    if axis_v.Length < 1e-9:
        axis_v = App.Vector(0, 0, 1)
    else:
        axis_v.normalize()

    obj.Placement = App.Placement(
        App.Vector(float(px), float(py), float(pz)),
        App.Rotation(axis_v, ang)
    )

def make_cyl_at(x, z, diam):
    r = diam/2.0
    return Part.makeCylinder(r, CUT_LEN, App.Vector(x, FRONT_Y - 5.0, z), App.Vector(0, 1, 0))

# Right cluster: cross + center
rp = [
    (0.0, 0.0),
    (RIGHT_BTN_SPACING, 0.0),
    (-RIGHT_BTN_SPACING, 0.0),
    (0.0, RIGHT_BTN_SPACING),
    (0.0, -RIGHT_BTN_SPACING),
]
right_cyls = [make_cyl_at(right_bezel_x + dx, lcd_zc + dz, RIGHT_BTN_DIAM) for dx, dz in rp]

# Left pair: two holes vertical
left_zs = [
    lcd_zc + LEFT_BTN_Z_OFFSET + (LEFT_BTN_SPACING/2.0),
    lcd_zc + LEFT_BTN_Z_OFFSET - (LEFT_BTN_SPACING/2.0),
]
left_cyls = [make_cyl_at(left_bezel_x, z, LEFT_BTN_DIAM) for z in left_zs]

all_btn_cuts = Part.makeCompound(right_cyls + left_cyls)

    did_cut = False
    for target in (hb, hf):
        tbb = target.Shape.BoundBox
        if bbox_intersects(sd_bb, tbb):
            try:
                target.Shape = remove_splitters(target.Shape.cut(sd_world))
                did_cut = True

                print("SD cut failed on", target.Name, "->", e)

        doc.removeObject(sd.Name)
    except:
        pass

def make_L_wall(len_x, len_y, thk_xleg, thk_yleg, height):
    """
    L wall in LOCAL coords:
      - Leg X: length=len_x along +X, thickness=thk_xleg along +Y
      - Leg Y: length=len_y along +Y, thickness=thk_yleg along +X
    So each leg thickness is independently controllable.
    """
    legX = Part.makeBox(float(len_x), float(thk_xleg), float(height), App.Vector(0, 0, 0))
    legY = Part.makeBox(float(thk_yleg), float(len_y), float(height), App.Vector(0, 0, 0))
    return remove_splitters(legX.fuse(legY))

def make_rect_wall(len_x, thk_y, height):
    return remove_splitters(Part.makeBox(float(len_x), float(thk_y), float(height), App.Vector(0,0,0)))

def apply_wall_feature(name, shape, placement_dict):
    obj, created = ensure_feature(name)
    obj.Shape = shape

    # Apply placement either always, or only on creation
    if ALWAYS_APPLY_PLACEMENT or created:
        apply_gui_placement(obj, placement_dict)

    return obj
