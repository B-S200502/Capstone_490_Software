def must(p):
    if not os.path.exists(p):
        raise FileNotFoundError(f"Missing: {p}")

def load_shape(filename):
    p = os.path.join(BASE_DIR, filename)
    must(p)
    return Part.read(p)

def add_obj(doc, name, shape):
    o = doc.addObject("Part::Feature", name)
    o.Shape = shape
    return o

def apply_placement_about_origin(shape, base_mm, axis_xyz, angle_deg):
    s = shape.copy()
    s.rotate(App.Vector(0,0,0), App.Vector(*axis_xyz), angle_deg)
    s.translate(App.Vector(*base_mm))
    return s

def translate(shape, dx, dy, dz):
    s = shape.copy()
    s.translate(App.Vector(dx,dy,dz))
    return s

def bb_union(bbs):
    bb = bbs[0]
    out = App.BoundBox(bb.XMin, bb.YMin, bb.ZMin, bb.XMax, bb.YMax, bb.ZMax)
    for b in bbs[1:]:
        out.add(b)
    return out

def clamp(v, lo, hi):
    return max(lo, min(hi, v))

def remove_splitters(shape):
    try:
        return shape.removeSplitter()
    except:
        return shape

# 4) Rounded-rect face in XZ extruded along Y
# ==========================================================
def rounded_rect_face_xz(xL, xR, zB, zT, r, y_at):
    w = xR - xL
    h = zT - zB
    r = max(0.0, min(r, w/2.0, h/2.0))
    if r < 1e-6:
        p1 = App.Vector(xL, y_at, zB)
        p2 = App.Vector(xR, y_at, zB)
        p3 = App.Vector(xR, y_at, zT)
        p4 = App.Vector(xL, y_at, zT)
        wire = Part.Wire([Part.makeLine(p1,p2),Part.makeLine(p2,p3),Part.makeLine(p3,p4),Part.makeLine(p4,p1)])
        return Part.Face(wire)

    rt = r / math.sqrt(2.0)
    topL = App.Vector(xL + r, y_at, zT)
    topR = App.Vector(xR - r, y_at, zT)
    rightT = App.Vector(xR, y_at, zT - r)
    rightB = App.Vector(xR, y_at, zB + r)
    botR = App.Vector(xR - r, y_at, zB)
    botL = App.Vector(xL + r, y_at, zB)
    leftB = App.Vector(xL, y_at, zB + r)
    leftT = App.Vector(xL, y_at, zT - r)

    tr_mid = App.Vector((xR - r) + rt, y_at, (zT - r) + rt)
    br_mid = App.Vector((xR - r) + rt, y_at, (zB + r) - rt)
    bl_mid = App.Vector((xL + r) - rt, y_at, (zB + r) - rt)
    tl_mid = App.Vector((xL + r) - rt, y_at, (zT - r) + rt)

    edges = [
        Part.makeLine(topL, topR),
        Part.Arc(topR, tr_mid, rightT).toShape(),
        Part.makeLine(rightT, rightB),
        Part.Arc(rightB, br_mid, botR).toShape(),
        Part.makeLine(botR, botL),
        Part.Arc(botL, bl_mid, leftB).toShape(),
        Part.makeLine(leftB, leftT),
        Part.Arc(leftT, tl_mid, topL).toShape(),
    ]
    return Part.Face(Part.Wire(edges))

# 5) Port tunnel cut (sample part edge region)
# ==========================================================
def cut_port_tunnel(shell, shape, wall_x, side, front_y, back_y):
    bb = shape.BoundBox
    if side.upper() == "RIGHT":
        edge_x = bb.XMax
        slab = Part.makeBox(PORT_SLAB_THICK, (bb.YLength + 40), (bb.ZLength + 40),
                            App.Vector(edge_x - PORT_SLAB_THICK, bb.YMin - 20, bb.ZMin - 20))
    else:
        edge_x = bb.XMin
        slab = Part.makeBox(PORT_SLAB_THICK, (bb.YLength + 40), (bb.ZLength + 40),
                            App.Vector(edge_x, bb.YMin - 20, bb.ZMin - 20))

    try:
        common = shape.common(slab)
        cbb = common.BoundBox
        y0 = cbb.YMin - PORT_MARGIN_Y
        y1 = cbb.YMax + PORT_MARGIN_Y
        z0 = cbb.ZMin - PORT_MARGIN_Z
        z1 = cbb.ZMax + PORT_MARGIN_Z
    except:
        y0 = bb.YMin - PORT_MARGIN_Y
        y1 = bb.YMax + PORT_MARGIN_Y
        z0 = (bb.ZMin + 0.25*bb.ZLength) - PORT_MARGIN_Z
        z1 = (bb.ZMin + 0.85*bb.ZLength) + PORT_MARGIN_Z

    y0 = clamp(y0, front_y + 3.0, back_y - 3.0)
    y1 = clamp(y1, front_y + 3.0, back_y - 3.0)

    gap = abs(wall_x - edge_x)
    depth = max(PORT_MIN_DEPTH, gap + WALL + 10.0)

    if side.upper() == "RIGHT":
        port = Part.makeBox(depth, (y1-y0), (z1-z0), App.Vector(wall_x - depth, y0, z0))
    else:
        port = Part.makeBox(depth, (y1-y0), (z1-z0), App.Vector(wall_x, y0, z0))

    return shell.cut(port)
